Groups ERT variants into a single identifier. entry_domains absent_sites_from_json.update(lacking_domains) success_rate = (valid_snaps / total_snaps * 100) if total_snaps > 0 else 0 fully_available = (success_rate == 100.0) and not month_incomplete_flag records.append({ "YearMonth": month_key, "Year": int(year), "MonthNum": int(month), "Total": total_snaps, "Valid": valid_snaps, "Missing": missing_snaps, "Corrupt": corrupt_snaps, "MonthIncompleteFlag": month_incomplete_flag, "SuccessRate": round(success_rate, 2), "FullyAvailable": fully_available, "MissingSites": ", ".join(sorted(missing_sites_on_disk)) if missing_sites_on_disk else "None", "CorruptSites": ", ".join(sorted(corrupt_sites_on_disk)) if corrupt_sites_on_disk else "None", "AbsentSites": ", ".join(sorted(absent_sites_from_json)) if absent_sites_from_json else "None" }) if not fully_available: problematic_report[month_key] = { "month_incomplete_flag": month_incomplete_flag, "lacking_domains_in_json": sorted(list(absent_sites_from_json)), "missing_snapshots_on_disk": detailed_missing, "corrupt_snapshots_on_disk": detailed_corrupt } df = pd.DataFrame(records) if df.empty: return # Save the detailed JSON report report_path = os.path.join(OUTPUT_DIR, REPORT_FILE) with open(report_path, "w", encoding="utf-8") as f: json.dump(problematic_report, f, indent=4) print(f"[✓] Detailed Problematic Snapshots Report saved to: {report_path}") # ===================================================================== # FIGURE 2: PROBLEMATIC BREAKDOWN (Bars & Incomplete Flags Only) # ===================================================================== prob_df = df[~df["FullyAvailable"]].sort_values(["Year", "MonthNum"]).copy() if not prob_df.empty: prob_df['HoverMissing'] = prob_df.apply( lambda r: f"Disk Missing: {r['Missing']}
